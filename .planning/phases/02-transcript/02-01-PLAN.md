    """
    audio_extractor.py
    ~~~~~~~~~~~~~~~~~~
    Extract 16kHz mono PCM audio from a video for downstream whisper.cpp
    transcription. Uses ffmpeg (already a required dependency of LOCALOCR).
    """
    import json
    import logging
    from pathlib import Path

    from src.common.subprocess_utils import (
        require_binary, run_subprocess,
        BinaryNotFoundError, SubprocessError,
    )

    logger = logging.getLogger(__name__)

    class AudioExtractionError(Exception): ...
    class NoAudioStreamError(AudioExtractionError):
        """Raised when the video contains no audio stream at all."""

    def _has_audio_stream(video: Path, ffprobe_bin: str) -> bool:
        cmd = [ffprobe_bin, "-v", "quiet", "-print_format", "json",
               "-show_streams", "-select_streams", "a", str(video)]
        stderr = run_subprocess(cmd, str(video), timeout=30, capture_stderr=True)
        # ffprobe writes streams JSON to stdout, not stderr — use a plain check
        # by re-running without capture_stderr to get stdout:
        # (Simplest safe impl: call subprocess.run directly since this is a
        # one-off — OR extend run_subprocess to return (stdout, stderr).)
        # For this plan: extend run_subprocess to also expose stdout.
        ...

    def extract_audio(video_path: str, output_wav_path: str) -> Path:
        """
        Extract audio track to 16kHz mono PCM WAV.

        Raises:
            NoAudioStreamError — video has no audio track
            AudioExtractionError — ffmpeg failed
            BinaryNotFoundError — ffmpeg missing (via require_binary)
        """
        video = Path(video_path).resolve()
        if not video.is_file():
            raise AudioExtractionError(f"Video not found: {video}")

        ffmpeg_bin = require_binary("ffmpeg")
        ffprobe_bin = require_binary("ffprobe")

        if not _has_audio_stream(video, ffprobe_bin):
            raise NoAudioStreamError(
                f"No audio stream in {video} — skipping transcription"
            )

        out = Path(output_wav_path).resolve()
        out.parent.mkdir(parents=True, exist_ok=True)

        cmd = [
            ffmpeg_bin, "-hide_banner", "-loglevel", "error",
            "-y",  # overwrite
            "-i", str(video),
            "-vn",  # no video
            "-acodec", "pcm_s16le",
            "-ar", "16000",  # 16 kHz
            "-ac", "1",      # mono
            str(out),
        ]
        run_subprocess(cmd, str(video), timeout=600)
        logger.info("Extracted audio: %s", out)
        return out

**Note on `_has_audio_stream`:** during implementation, either extend `run_subprocess` to also return stdout (cleanest — preferred), OR use a small direct `subprocess.run` here with a comment explaining why. Prefer extending `run_subprocess` since audio-stream probing is a legitimate need shared by any future audio-consuming module.

Update `run_subprocess` signature to `-> tuple[str | None, str | None]` (stdout, stderr) when either capture flag is set. Add a `capture_stdout: bool = False` param. Update the frame_extractor call site (only one place uses this — no cascading changes).
