    """
    whisper_transcriber.py
    ~~~~~~~~~~~~~~~~~~~~~~
    Subprocess wrapper around whisper.cpp `whisper-cli`. Runs local
    transcription and returns normalized Segment records.
    """
    import json
    import logging
    from dataclasses import dataclass, asdict
    from pathlib import Path
    from typing import Optional

    from src.common.subprocess_utils import (
        require_binary, run_subprocess,
        BinaryNotFoundError, SubprocessError,
    )

    logger = logging.getLogger(__name__)

    class WhisperNotAvailableError(Exception):
        """Whisper binary or model not found; caller should skip transcription."""
    class WhisperFailureError(Exception):
        """Whisper ran but failed to produce usable output."""

    @dataclass(frozen=True)
    class Segment:
        start: float
        end: float
        text: str
        speaker: Optional[str] = None

    _DEFAULT_BINARY_CANDIDATES = ("whisper-cli", "main", "whisper")

    def _locate_binary(configured: str | None) -> str:
        """Try configured name first, then known aliases. Raise if none found."""
        candidates = [configured] if configured else []
        candidates.extend(c for c in _DEFAULT_BINARY_CANDIDATES if c not in candidates)
        for name in candidates:
            try:
                return require_binary(name)
            except BinaryNotFoundError:
                continue
        raise WhisperNotAvailableError(
            "whisper.cpp binary not found. Install via `brew install whisper-cpp` "
            "and set transcript_config.binary if your binary uses a non-standard name."
        )

    def _locate_model(cfg: dict) -> Path:
        model_dir = Path(cfg.get("model_dir", "~/.whisper.cpp/models")).expanduser()
        model_name = cfg.get("model", "base.en")
        candidate = model_dir / f"ggml-{model_name}.bin"
        if not candidate.is_file():
            raise WhisperNotAvailableError(
                f"whisper model not found: {candidate}. "
                f"Download via: bash download-ggml-model.sh {model_name}"
            )
        return candidate

    def _parse_whisper_json(json_path: Path) -> list[Segment]:
        with open(json_path, encoding="utf-8") as f:
            data = json.load(f)
        raw_segments = data.get("transcription", [])
        segments = []
        for s in raw_segments:
            offsets = s.get("offsets", {})
            start_ms = offsets.get("from", 0)
            end_ms = offsets.get("to", 0)
            text = (s.get("text") or "").strip()
            if not text:
                continue
            segments.append(Segment(
                start=start_ms / 1000.0,
                end=end_ms / 1000.0,
                text=text,
                speaker=None,
            ))
        segments.sort(key=lambda seg: seg.start)
        return segments

    def transcribe(audio_path: str, cfg: dict) -> list[Segment]:
        """
        Transcribe an audio file with whisper.cpp.

        Raises:
            WhisperNotAvailableError — binary or model missing (caller should skip)
            WhisperFailureError — subprocess ran but output unreadable
        """
        audio = Path(audio_path).resolve()
        if not audio.is_file():
            raise WhisperFailureError(f"Audio file not found: {audio}")

        binary = _locate_binary(cfg.get("binary"))
        model = _locate_model(cfg)
        language = cfg.get("language", "en")
        threads = int(cfg.get("threads", 8))

        output_prefix = audio.with_suffix("")  # e.g. /tmp/x.wav -> /tmp/x
        json_out = output_prefix.with_suffix(".json")

        cmd = [
            binary,
            "-m", str(model),
            "-f", str(audio),
            "-oj",  # JSON output
            "-of", str(output_prefix),
            "-l", language,
            "-t", str(threads),
        ]
        # Long-running: whisper on base.en typically 5-10x realtime on M-series
        timeout = 3600  # 1h safety cap; adjust if we see it hit
        try:
            run_subprocess(cmd, str(audio), timeout=timeout)
        except SubprocessError as exc:
            raise WhisperFailureError(f"whisper-cli failed: {exc}") from exc

        if not json_out.is_file():
            raise WhisperFailureError(f"whisper produced no JSON output at {json_out}")

        return _parse_whisper_json(json_out)

    """
    correlator.py
    ~~~~~~~~~~~~~
    Pure functions that map frame timestamps to surrounding transcript
    context. No I/O, no side effects — trivially unit-testable.
    """
    import re
    from bisect import bisect_left
    from typing import Iterable
    from collections import Counter

    from .whisper_transcriber import Segment

    _TS_RE = re.compile(r"^(\d{2})m(\d{2})s$")

    def frame_timestamp_seconds(frame: dict) -> float:
        """Parse the 'XXmYYs' timestamp field of a frame dict."""
        ts = str(frame.get("timestamp", "")).strip()
        m = _TS_RE.match(ts)
        if not m:
            return 0.0
        return int(m.group(1)) * 60 + int(m.group(2))

    def correlate(
        frame_ts: float,
        segments: list[Segment],
        window_seconds: float,
    ) -> dict:
        """
        Return the transcript window around `frame_ts`.

        Structure:
            {
                "before":  "concatenated text of segments ending before frame_ts",
                "at":      "concatenated text of segments containing frame_ts",
                "after":   "concatenated text of segments starting after frame_ts",
                "speaker": <majority speaker in window, or None>,
            }
        """
        if not segments:
            return {"before": "", "at": "", "after": "", "speaker": None}

        window_start = frame_ts - window_seconds
        window_end   = frame_ts + window_seconds

        # Only segments overlapping the window
        in_window = [
            s for s in segments
            if s.end >= window_start and s.start <= window_end
        ]

        before, at, after = [], [], []
        for s in in_window:
            if s.end < frame_ts:
                before.append(s.text)
            elif s.start > frame_ts:
                after.append(s.text)
            else:
                # segment contains frame_ts
                at.append(s.text)

        speakers = [s.speaker for s in in_window if s.speaker]
        speaker = Counter(speakers).most_common(1)[0][0] if speakers else None

        return {
            "before":  " ".join(before).strip(),
            "at":      " ".join(at).strip(),
            "after":   " ".join(after).strip(),
            "speaker": speaker,
        }

    def enrich_ocr_results(
        results: list[dict],
        segments: list[Segment],
        window_seconds: float,
    ) -> list[dict]:
        """
        Return a new list where each MATCHED result has a `transcript_context`
        field. Unmatched results are returned unchanged.
        """
        if not segments:
            return list(results)
        out = []
        for r in results:
            if not r.get("matched"):
                out.append(r)
                continue
            new = dict(r)
            new["transcript_context"] = correlate(
                frame_timestamp_seconds(r),
                segments,
                window_seconds,
            )
            out.append(new)
        return out
