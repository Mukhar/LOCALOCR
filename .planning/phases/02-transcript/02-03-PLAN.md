Create `src/transcript/pipeline_glue.py`:

    """
    pipeline_glue.py
    ~~~~~~~~~~~~~~~~
    Glue that runs audio_extractor + whisper_transcriber in a background
    thread so the main pipeline (OCR + match + organize) can proceed in
    parallel. Handles every failure mode from the transcript degradation
    matrix and NEVER raises.
    """
    import json
    import logging
    from concurrent.futures import Future, ThreadPoolExecutor
    from pathlib import Path
    from typing import Optional

    from .audio_extractor import (
        extract_audio, NoAudioStreamError, AudioExtractionError,
    )
    from .whisper_transcriber import (
        transcribe, Segment,
        WhisperNotAvailableError, WhisperFailureError,
    )

    logger = logging.getLogger(__name__)

    _EXECUTOR = ThreadPoolExecutor(max_workers=1, thread_name_prefix="whisper")

    def _run_transcription(video_path: str, cfg: dict, metadata_dir: Path) -> Optional[list[Segment]]:
        """Do the work: extract audio → whisper → write transcript.json.
        Returns None on any graceful failure."""
        try:
            wav = metadata_dir / "_audio.wav"
            extract_audio(video_path, str(wav))
        except NoAudioStreamError as exc:
            logger.warning("Transcription skipped: %s", exc)
            return None
        except AudioExtractionError as exc:
            logger.warning("Transcription skipped (audio extraction failed): %s", exc)
            return None

        try:
            segments = transcribe(str(wav), cfg)
        except WhisperNotAvailableError as exc:
            logger.warning("Transcription skipped: %s", exc)
            return None
        except WhisperFailureError as exc:
            logger.warning("Transcription skipped (whisper failed): %s", exc)
            return None
        finally:
            wav.unlink(missing_ok=True)

        # Persist
        transcript_path = metadata_dir / "transcript.json"
        with open(transcript_path, "w", encoding="utf-8") as f:
            json.dump(
                [{"start": s.start, "end": s.end, "text": s.text,
                  "speaker": s.speaker} for s in segments],
                f, indent=2, ensure_ascii=False,
            )
        logger.info("Transcript saved: %s (%d segments)",
                    transcript_path, len(segments))
        return segments

    def kickoff_transcription(
        video_path: str,
        cfg: dict,
        metadata_dir: Path,
    ) -> Optional[Future]:
        """Schedule transcription in background. Returns Future, or None if disabled."""
        if not cfg.get("enabled", False):
            logger.info("Transcription disabled (transcript_config.enabled=false)")
            return None
        return _EXECUTOR.submit(_run_transcription, video_path, cfg, metadata_dir)
