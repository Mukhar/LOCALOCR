    from fastapi import APIRouter, Request, Form, HTTPException
    from fastapi.responses import RedirectResponse, HTMLResponse
    from fastapi.templating import Jinja2Templates
    import json
    from pathlib import Path

    from ..services.db import list_runs, get_run, list_picks, delete_run
    from ..services.run_manager import get_run_manager

    router = APIRouter()
    templates = Jinja2Templates(directory="src/web/templates")

    @router.get("/", include_in_schema=False)
    def root():
        return RedirectResponse(url="/runs")

    @router.get("/runs", response_class=HTMLResponse)
    def runs_list(request: Request):
        db_path = request.app.state.db_path
        runs = list_runs(db_path)
        return templates.TemplateResponse("runs_list.html",
            {"request": request, "runs": runs})

    @router.get("/runs/new", response_class=HTMLResponse)
    def run_new_form(request: Request):
        from .videos import _list_input_videos
        return templates.TemplateResponse("run_new.html",
            {"request": request, "videos": _list_input_videos()})

    @router.post("/runs")
    def run_new_submit(request: Request,
                       video_path: str = Form(...),
                       config_profile: str = Form("default")):
        db_path = request.app.state.db_path
        cfg_file = {"default": "config/config.json",
                    "scene": "config/config.scene.example.json",
                    "hybrid": "config/config.hybrid.example.json",
                    "transcript": "config/config.transcript.example.json"}.get(
                        config_profile, "config/config.json")
        with open(cfg_file) as f:
            config = json.load(f)
        config["video_path"] = video_path
        manager = get_run_manager(db_path)
        run_id = manager.start_run(config)
        return RedirectResponse(url=f"/runs/{run_id}", status_code=303)

    @router.get("/runs/{run_id}", response_class=HTMLResponse)
    def run_detail(request: Request, run_id: int):
        db_path = request.app.state.db_path
        run = get_run(db_path, run_id)
        if not run:
            raise HTTPException(status_code=404, detail="Run not found")
        picks = list_picks(db_path, run_id)
        return templates.TemplateResponse("run_detail.html",
            {"request": request, "run": run, "picks": picks})

    @router.delete("/runs/{run_id}")
    def run_delete(request: Request, run_id: int):
        db_path = request.app.state.db_path
        delete_run(db_path, run_id)
        return HTMLResponse("")  # HTMX removes the row on 200

    from fastapi import APIRouter, Request, UploadFile, File, HTTPException
    from fastapi.responses import FileResponse, JSONResponse
    from pathlib import Path

    router = APIRouter()

    _INPUT_DIR = Path("input_videos").resolve()
    _SUPPORTED = {".mp4", ".webm", ".mkv", ".mov", ".m4v", ".avi"}
    _MAX_UPLOAD_MB = 500

    def _list_input_videos() -> list[dict]:
        if not _INPUT_DIR.exists():
            return []
        return [{"name": p.name, "path": str(p)}
                for p in sorted(_INPUT_DIR.iterdir())
                if p.is_file() and p.suffix.lower() in _SUPPORTED]

    @router.get("/videos")
    def list_videos():
        return _list_input_videos()

    @router.post("/videos/upload")
    async def upload_video(file: UploadFile = File(...)):
        # Path-traversal defense
        safe_name = Path(file.filename or "upload.mp4").name
        if Path(safe_name).suffix.lower() not in _SUPPORTED:
            raise HTTPException(400, f"Unsupported extension: {safe_name}")
        _INPUT_DIR.mkdir(parents=True, exist_ok=True)
        dest = _INPUT_DIR / safe_name

        # Size cap
        max_bytes = _MAX_UPLOAD_MB * 1024 * 1024
        written = 0
        with open(dest, "wb") as f:
            while chunk := await file.read(1024 * 1024):
                written += len(chunk)
                if written > max_bytes:
                    f.close()
                    dest.unlink(missing_ok=True)
                    raise HTTPException(413, f"Upload exceeds {_MAX_UPLOAD_MB}MB")
                f.write(chunk)
        return JSONResponse({"name": safe_name, "size": written})

    @router.get("/videos/{name}")
    def get_video(name: str):
        safe = Path(name).name  # strip any traversal
        p = _INPUT_DIR / safe
        if not p.is_file():
            raise HTTPException(404, "Video not found")
        return FileResponse(p)  # FastAPI handles byte-range automatically
  </action>

    from fastapi import FastAPI
    from fastapi.staticfiles import StaticFiles
    from pathlib import Path

    from .services.db import init_db
    from .routes import runs, videos

    def create_app(db_path: Path | None = None) -> FastAPI:
        app = FastAPI(title="LOCALOCR", version="1.1")
        app.state.db_path = db_path or Path("output/localocr.sqlite").resolve()
        init_db(app.state.db_path)

        app.include_router(runs.router)
        app.include_router(videos.router)

        # Serve output/ for frame thumbnails
        output_dir = Path("output").resolve()
        if output_dir.exists():
            app.mount("/output", StaticFiles(directory=output_dir), name="output")

        # Serve static/ for JS/CSS added in 03-04
        static_dir = Path("src/web/static").resolve()
        if static_dir.exists():
            app.mount("/static", StaticFiles(directory=static_dir), name="static")

        @app.get("/health")
        def health():
            return {"status": "ok", "db_path": str(app.state.db_path)}

        return app
  </action>
