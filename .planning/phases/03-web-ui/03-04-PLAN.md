    """
    Server-Sent Events endpoint for live run progress.
    """
    import json
    from fastapi import APIRouter, Request
    from fastapi.responses import HTMLResponse
    from fastapi.templating import Jinja2Templates
    from sse_starlette.sse import EventSourceResponse

    from ..services.event_bus import event_bus
    from ..services.db import get_run

    router = APIRouter()
    templates = Jinja2Templates(directory="src/web/templates")

    @router.get("/runs/{run_id}/events")
    async def stream_events(request: Request, run_id: int):
        async def event_generator():
            async for event in event_bus.subscribe(run_id):
                if await request.is_disconnected():
                    break
                # SSE clients (HTMX sse ext) render event data as HTML snippets
                # For simplicity, emit JSON strings and let HTMX swap templates.
                yield {"event": event.get("type", "message"),
                       "data": json.dumps(event)}
        return EventSourceResponse(event_generator())

    @router.get("/runs/{run_id}/progress", response_class=HTMLResponse)
    def progress_panel(request: Request, run_id: int):
        """Initial HTML for the progress panel — HTMX SSE then updates parts."""
        db_path = request.app.state.db_path
        run = get_run(db_path, run_id)
        return templates.TemplateResponse("run_progress.html",
            {"request": request, "run": run})

Add before `</section>`:

    document.body.addEventListener('htmx:sseMessage', (evt) => {
      try {
        const data = JSON.parse(evt.detail.data);
        if (data.type === 'log') {
          const line = document.createElement('div');
          line.textContent = `[${data.level}] ${data.message}`;
          document.getElementById('run-log').appendChild(line);

Create `src/web/static/seek.js`:

    /*
     * Click any element with data-seek="<seconds>" → seek the #source-video
     * <video> element to that time and play. Delegated listener so it works
     * for elements added later via HTMX swaps.
     */
    document.addEventListener("click", (e) => {
      const el = e.target.closest("[data-seek]");
      if (!el) return;
      const video = document.getElementById("source-video");
      if (!video) return;
      const seconds = parseFloat(el.dataset.seek);
      if (Number.isNaN(seconds)) return;
      video.currentTime = seconds;
      const p = video.play();
      if (p && typeof p.catch === "function") p.catch(() => {});
      video.scrollIntoView({ behavior: "smooth", block: "center" });
    });
