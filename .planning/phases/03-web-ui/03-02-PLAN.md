    """
    Entry point for pipeline subprocesses spawned by RunManager.

    Emits structured JSON events to stdout so the parent can parse and
    republish. Called via:
        python -m src.web.services.runner_subprocess --run-id 42 --config <json>
    """
    import argparse
    import json
    import logging
    import sys
    from pathlib import Path

    from src.pipeline.pipeline_runner import run_pipeline

    class StructuredEventHandler(logging.Handler):
        """Emit each log record as a JSON line to stdout."""
        def __init__(self, run_id: int):
            super().__init__()
            self.run_id = run_id

        def emit(self, record):
            event = {
                "type": "log",
                "level": record.levelname,
                "message": self.format(record),
                "logger": record.name,
            }
            sys.stdout.write(json.dumps(event) + "\n")
            sys.stdout.flush()

    def _emit(event: dict) -> None:
        sys.stdout.write(json.dumps(event) + "\n")
        sys.stdout.flush()

    def main() -> int:
        parser = argparse.ArgumentParser()
        parser.add_argument("--run-id", type=int, required=True)
        parser.add_argument("--config", type=str, required=True)
        args = parser.parse_args()

        # Structured logging → stdout
        handler = StructuredEventHandler(args.run_id)
        handler.setFormatter(logging.Formatter("%(message)s"))
        root = logging.getLogger()
        root.setLevel(logging.INFO)
        root.addHandler(handler)

        config = json.loads(args.config)
        _emit({"type": "start", "run_id": args.run_id})
        try:
            summary = run_pipeline(config)
            _emit({"type": "summary", "summary": summary})
            _emit({"type": "done", "status": "completed"})
            return 0
        except KeyboardInterrupt:
            _emit({"type": "done", "status": "interrupted"})
            return 130
        except Exception as exc:  # noqa: BLE001
            _emit({"type": "error", "message": str(exc)})
            _emit({"type": "done", "status": "failed"})
            return 1

    if __name__ == "__main__":
        raise SystemExit(main())

    """
    Spawns pipeline subprocesses and pumps their structured event streams
    into the DB + event bus.
    """
    import json
    import logging
    import subprocess
    import sys
    import threading
    from pathlib import Path
    from queue import Queue
    from typing import Optional

    from .db import insert_run, finalize_run
    from .event_bus import event_bus

    logger = logging.getLogger(__name__)

    class RunManager:
        def __init__(self, db_path: Path, max_concurrent: int = 1):
            self.db_path = db_path
            self.max_concurrent = max_concurrent
            self._active: dict[int, subprocess.Popen] = {}
            self._pending: Queue = Queue()
            self._lock = threading.Lock()

        def start_run(self, config: dict) -> int:
            video_path = config.get("video_path", "")
            run_id = insert_run(self.db_path, video_path, config)
            with self._lock:
                if len(self._active) >= self.max_concurrent:
                    self._pending.put((run_id, config))
                    event_bus.publish(run_id, {
                        "type": "queued",
                        "queued_position": self._pending.qsize(),
                    })
                    return run_id
            self._spawn(run_id, config)
            return run_id

        def _spawn(self, run_id: int, config: dict) -> None:
            cmd = [sys.executable, "-m", "src.web.services.runner_subprocess",
                   "--run-id", str(run_id), "--config", json.dumps(config)]
            proc = subprocess.Popen(
                cmd,
                stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                text=True, bufsize=1,
            )
            with self._lock:
                self._active[run_id] = proc
            threading.Thread(
                target=self._pump, args=(run_id, proc),
                daemon=True, name=f"pump-{run_id}",
            ).start()

        def _pump(self, run_id: int, proc: subprocess.Popen) -> None:
            summary: Optional[dict] = None
            final_status = "failed"
            try:
                for line in proc.stdout:
                    line = line.rstrip()
                    if not line:
                        continue
                    try:
                        event = json.loads(line)
                    except json.JSONDecodeError:
                        # Not a structured event — treat as log
                        event = {"type": "log", "level": "INFO", "message": line}
                    event_bus.publish(run_id, event)
                    if event.get("type") == "summary":
                        summary = event.get("summary")
                    if event.get("type") == "done":
                        final_status = event.get("status", "completed")
                proc.wait()
            finally:
                finalize_run(self.db_path, run_id, final_status, summary)
                event_bus.publish(run_id, {"type": "done", "status": final_status})
                with self._lock:
                    self._active.pop(run_id, None)
                self._drain_pending()

        def _drain_pending(self) -> None:
            with self._lock:
                if self._active or self._pending.empty():
                    return
            try:
                run_id, config = self._pending.get_nowait()
            except Exception:
                return
            self._spawn(run_id, config)

        def cancel_run(self, run_id: int) -> bool:
            with self._lock:
                proc = self._active.get(run_id)
            if proc:
                proc.terminate()
                return True
            return False

    # Module-level singleton bound at app-creation time
    _instance: Optional[RunManager] = None

    def get_run_manager(db_path: Path) -> RunManager:
        global _instance
        if _instance is None:
            _instance = RunManager(db_path)
        return _instance
  </action>
